dat <- dat %>%
  mutate(
    text_clean = comment_text,
    text_clean = str_replace_all(text_clean, "http[s]?://\\S+", " "),   # URLs
    text_clean = str_replace_all(text_clean, "@\\w+", " "),             # mentions
    text_clean = str_replace_all(text_clean, "#\\w+", " "),             # hashtags
    text_clean = str_replace_all(text_clean, "[^\\x01-\\x7F]", " "),    # emoji / non-ASCII
    text_clean = str_to_lower(text_clean),
    text_clean = str_replace_all(text_clean, "[[:digit:]]+", " "),
    text_clean = str_replace_all(text_clean, "[[:punct:]]", " "),
    text_clean = str_squish(text_clean),                                # collapse spaces

bing <- get_sentiments("bing")     # bundled with tidytext, works offline

comment_scores <- tokens %>%
  inner_join(bing, by = "word") %>%
  count(comment_id, sentiment) %>%
  pivot_wider(names_from = sentiment, values_from = n, values_fill = 0) %>%
  mutate(net_score = positive - negative)

dat <- dat %>%
  left_join(comment_scores, by = "comment_id") %>%
  mutate(
    positive  = replace_na(positive, 0),
    negative  = replace_na(negative, 0),
    net_score = replace_na(net_score, 0),
    pred_label = factor(case_when(net_score > 0 ~ "Positive",
                                  net_score < 0 ~ "Negative",
                                  TRUE          ~ "Neutral"),
                        levels = c("Negative", "Neutral", "Positive"))

# Per-class precision, recall and F1
perf <- tibble(class = colnames(cm)) %>%
  rowwise() %>%
  mutate(
    TP        = cm[class, class],
    precision = TP / sum(cm[class, ]),
    recall    = TP / sum(cm[, class]),
    f1        = 2 * precision * recall / (precision + recall)
  ) %>%
  ungroup() %>%
  mutate(across(precision:f1, ~ round(.x, 3)))

7. WHY THE LEXICON FAILS: NEGATION AND SARCASM

## ---------------------------------------------------------------------------
negators <- c("not", "no", "never", "without", "hardly", "zero", "nothing", "cannot")

bigrams <- dat %>%
  select(comment_id, text_clean) %>%
  unnest_tokens(bigram, text_clean, token = "ngrams", n = 2) %>%
  separate(bigram, c("w1", "w2"), sep = " ")

negated_terms <- bigrams %>%
  filter(w1 %in% negators) %>%
  inner_join(bing, by = c("w2" = "word")) %>%
  count(w1, w2, sentiment, sort = TRUE)
negated_terms                   # every row here was scored with the wrong sign
# A tibble: 8 × 4

  <chr> <chr>      <chr>     <int>

5 no    regrets    negative      1

# Simple correction: flip the sign of any sentiment word preceded by a negator
correction <- bigrams %>%
  filter(w1 %in% negators) %>%
  inner_join(bing, by = c("w2" = "word")) %>%
  mutate(adj = if_else(sentiment == "positive", -2, 2)) %>%
  group_by(comment_id) %>%
  summarise(adjustment = sum(adj), .groups = "drop")

dat <- dat %>%
  left_join(correction, by = "comment_id") %>%
  mutate(adjustment  = replace_na(adjustment, 0),
         score_adj   = net_score + adjustment,
         pred_adj    = factor(case_when(score_adj > 0 ~ "Positive",
                                        score_adj < 0 ~ "Negative",
                                        TRUE          ~ "Neutral"),
                              levels = c("Negative", "Neutral", "Positive")))

cm2 <- table(Predicted = dat$pred_adj, Actual = dat$human_label)
